// State your players can see
#[derive(HearthState, Default)]
pub struct Match {
    #[sync]
    pub score: i64,
}

#[hearth_mp::room(mode = "event")]
impl Room for Counter {
    type State = Match;
    type Intent = Intent;

    // Every action stays authoritative
    async fn on_intent(
        &mut self,
        ctx: &mut Ctx<Self>,
        _who: &Client,
        intent: Self::Intent,
    ) -> hearth_mp::Result<()> {
        match intent {
            Intent::Add(n) => ctx.state().score += n,
        }
        Ok(())
    }
}
